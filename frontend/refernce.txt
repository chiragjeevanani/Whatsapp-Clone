        // Move focus logic for OTP inputs
        function moveToNext(input, event) {
            if (input.value.length > 1) {
                input.value = input.value.slice(0, 1);
            }
            if (input.value.length === 1 && input.nextElementSibling) {
                input.nextElementSibling.focus();
            }
        }

        // Handle backspace to move to previous input
        document.getElementById('otp-container').addEventListener('keydown', function(e) {
            if (e.key === 'Backspace' && e.target.value === '' && e.target.previousElementSibling) {
                e.target.previousElementSibling.focus();
            }
        });

        // Simple Timer Logic
        let timeLeft = 59;
        let timerId;

        function startTimer() {
            timeLeft = 59;
            document.getElementById('timer-text').classList.remove('hidden');
            document.getElementById('resend-btn').classList.add('hidden');
            
            clearInterval(timerId);
            timerId = setInterval(() => {
                if (timeLeft <= 0) {
                    clearInterval(timerId);
                    document.getElementById('timer-text').classList.add('hidden');
                    document.getElementById('resend-btn').classList.remove('hidden');
                } else {
                    document.getElementById('timer').innerText = `0:${timeLeft < 10 ? '0' : ''}${timeLeft}`;
                }
                timeLeft -= 1;
            }, 1000);
        }

        // Start timer on load
        window.onload = startTimer;

        // Mock verification
        function verifyOTP() {
            const inputs = document.querySelectorAll('.otp-input');
            let code = '';
            inputs.forEach(i => code += i.value);
            
            if(code.length === 6) {
                const btn = document.querySelector('button[type="submit"]');
                const icon = document.getElementById('verify-icon');
                
                // Success state simulation
                btn.classList.replace('bg-primary-container', 'bg-secondary');
                btn.classList.replace('hover:bg-primary', 'hover:bg-secondary');
                btn.querySelector('span:first-child').innerText = 'Verified';
                icon.innerText = 'check_circle';
                icon.setAttribute('data-icon', 'check_circle');
                
                // Optional success animation class could be added here
            }
        }
